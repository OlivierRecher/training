% F8 -- What actually correlates with accuracy.
\begin{figure}[!htbp]
\centering
\replegend{\swatch{clmodel}~1{,}173-user set \qquad \swatch{clink}~100 held-out users, \smob}
\fitwidth{%
\begin{tikzpicture}
\begin{axis}[rephbar, width=0.80\linewidth, height=5.4cm,
  title={Pearson correlation between a per-user feature, one per row, and that user's accuracy},
  title style={font=\footnotesize,yshift=1pt},
  xmin=-1.2, xmax=1.2, xtick={-1,-0.5,0,0.5,1},
  xlabel={Pearson $r$}, /pgf/bar width=5.5pt,
  symbolic y coords={len,vol,uniq,irr,dom,ent,switch},
  ytick={len,vol,uniq,irr,dom,ent,switch},
  yticklabels={records in the raw sequence, predictions contributed,
               distinct cells visited, irregularity of the switches,
               share of the dominant cell, spread over cells (entropy),
               \textbf{cell-switch rate}},
  y tick label style={font=\scriptsize,align=right},
  nodes near coords,
  /pgf/number format/.cd, fixed, fixed zerofill, precision=3, /pgfplots/.cd,
  every node near coord/.append style={font=\tiny}]
\addplot[fill=clmodel,draw=none] coordinates {(0.054,len)(0.154,vol)(-0.175,uniq)(0.696,dom)(-0.811,ent)(-0.971,switch)};
\addplot[fill=clink,draw=none]   coordinates {(-0.101,vol)(-0.232,uniq)(-0.140,irr)(0.285,dom)(-0.773,switch)};
\end{axis}
\end{tikzpicture}}
\caption{Per-user correlations with accuracy}
\label{fig:concentration}
\figtext{%
One row per candidate feature, named on the left, and the
length of its bar is the Pearson correlation between that feature and the
user's own accuracy; zero means the feature says nothing about the score,
$\pm1$ that it says everything, and the sign says in which direction.
Dark blue bars are measured on the 1{,}173-user set and teal bars on the 100
held-out users of a different dataset; features with a single bar were
computed on one dataset only. The ranking is not close. The cell-switch rate
reaches $r = -0.971$ and is confirmed at $-0.773$ on the second corpus, while
normalised entropy ($-0.811$) and dominant-cell frequency ($+0.696$) are the
same information seen from other angles and everything else stays below
$0.24$. The $-0.971$ is close to arithmetic, for the reason given in the text,
and says as much about a model imitating the trivial rule as about the users;
the distribution, not the coefficient, is the finding.}
\end{figure}
